--- Desempenho de todas as turmas no primeiro semestre 
DROP VIEW IF EXISTS Desempenho_Geral CASCADE;
CREATE VIEW Desempenho_Geral AS 
	SELECT sigla_disciplina, codigo_turma, AVG(media_final) AS media FROM participa_em
		WHERE vigencia LIKE '%1'
		GROUP BY sigla_disciplina, codigo_turma  ;
		
--- Alunos que reprovaram 
DROP VIEW IF EXISTS Reprovados CASCADE;
CREATE VIEW Reprovados AS 
	SELECT nusp_aluno, sigla_disciplina, media_final FROM participa_em
	WHERE media_final < 5.0 OR frequencia < 70.0 AND vigencia = '2025-1';

--- Turmas oferecidas em 2º Semestre 
DROP VIEW IF EXISTS Turmas_Oferecidas_2SEMESTRE CASCADE;
CREATE VIEW Turmas_Oferecidas_2SEMESTRE AS 
	SELECT t.sigla_disciplina, t.codigo_turma, u.nome AS prof_responsavel 
	FROM leciona t JOIN usuario u ON t.nusp_professor = u.nusp
	WHERE vigencia LIKE '%2';

--- Historico_Alunos 
DROP VIEW IF EXISTS Historico_Aluno CASCADE;
CREATE VIEW Historico_Aluno AS SELECT U.nome, U.nusp,  AVG(media_final) AS media, AVG(frequencia) AS frequencia
	FROM usuario U JOIN participa_em p ON p.nusp_aluno = U.nusp
	GROUP BY U.nusp;

--- Vinculo_Professor

SELECT

    vigencia,
    COUNT(CASE WHEN media_final >= 5.0 AND frequencia >= 70.0 THEN 1 END) AS total_aprovados,
    COUNT(nusp_aluno) AS total_alunos,

    sigla_disciplina, vigencia

    vigencia DESC, taxa_aprovacao ASC;

--- 1.Atualizar a media_final de um aluno
CREATE OR REPLACE FUNCTION atualizar_nota_aluno(
    p_nusp_aluno INT, 
    p_sigla_disc VARCHAR,
    p_cod_turma INT,
    p_vigencia VARCHAR,
    p_media_final NUMERIC
) RETURNS VOID AS $$
BEGIN

    SET media_final = p_media_final --- Mudança/Inserção da média final 
    WHERE nusp_aluno = p_nusp_aluno --- Localizando pelas chaves NUSP e chaves de Turma

      AND codigo_turma = p_cod_turma
      AND vigencia = p_vigencia;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Registro de matrícula não encontrado.';
    END IF;
END;
$$ LANGUAGE plpgsql;

--- 3.Promover/Mudar o cargo de um professor (encerra o vínculo atual e cria um novo)
CREATE OR REPLACE FUNCTION promover_professor(
    p_nusp INT,
    p_novo_cargo VARCHAR
) RETURNS VOID AS $$
BEGIN
    --- Encerrar o vínculo de trabalho atual
    UPDATE vinculo_trabalho
    SET data_fim = CURRENT_DATE
    WHERE nusp_vinculo = p_nusp
      AND data_fim IS NULL;

    --- Inserir o novo vínculo
    INSERT INTO vinculo_trabalho (nusp_vinculo, id_instituto, cargo, data_inicio)
    SELECT
        p_nusp,
        id_instituto, -- Mantém o mesmo instituto
        p_novo_cargo, -- Coloca esse cargo 
        CURRENT_DATE
    FROM vinculo_trabalho
    WHERE nusp_vinculo = p_nusp
    ORDER BY data_fim DESC
    LIMIT 1;
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Professor não possui vínculo de trabalho anterior.';
    END IF;
END;
$$ LANGUAGE plpgsql;

-- 6. Gera um relatório detalhado da turma
CREATE OR REPLACE FUNCTION gerar_relatorio_turma(
    p_sigla_disc VARCHAR,
    p_cod_turma VARCHAR,
    p_vigencia VARCHAR
)
RETURNS TABLE (
    nusp_aluno INT,
    nome_aluno VARCHAR,
    media_final NUMERIC,
	frequencia NUMERIC,
    status VARCHAR
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        P.nusp_aluno,
        u.nome,
		p.frequencia,
        p.media_final,
        (CASE
            WHEN p.media_final IS NULL THEN 'Pendente'
            WHEN p.media_final >= 5.0 AND p.frequencia >70 THEN 'Aprovado'
            ELSE 'Reprovado'
        END)::VARCHAR(50) AS status
    FROM
        participa_em AS p
    JOIN
        usuario u ON p.nusp_aluno = u.nusp
    WHERE
        p.sigla_disciplina = p_sigla_disc
        AND p.codigo_turma = p_cod_turma
        AND p.vigencia = p_vigencia
    ORDER BY
        u.nome;
END;
$$ LANGUAGE plpgsql;

-- 7. Transfere um funcionário para outro instituto
CREATE OR REPLACE FUNCTION transferir_funcionario(
    p_nusp INT,
    p_novo_instituto_id INT
) RETURNS VOID AS $$
BEGIN
    -- Encerrar o vínculo de trabalho atual 
    UPDATE vinculo_trabalho
    SET data_fim = CURRENT_DATE
    WHERE nusp_associado = p_nusp
      AND data_fim IS NULL;

    -- Inserir o novo vínculo no novo instituto, mantendo o cargo
    INSERT INTO vinculo_trabalho (nusp_associado, id_instituto, cargo, data_inicio)
    SELECT
        p_nusp,
        p_novo_instituto_id, -- Novo Instituto
        vt.cargo,           -- Mantém o cargo
        CURRENT_DATE
    FROM
        vinculo_trabalho vt
    WHERE
        nusp_associado = p_nusp
    ORDER BY
        data_fim DESC
    LIMIT 1;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Funcionário não possui vínculo de trabalho anterior para transferência.';
    END IF;
END;
$$ LANGUAGE plpgsql;

-- 8. Retorna todas as aulas agendadas para um dia da semana em um instituto 
CREATE OR REPLACE FUNCTION aulas_por_dia_semana(
    p_dia_semana VARCHAR,
    p_vigencia VARCHAR,
    p_id_instituto INTEGER
)
RETURNS TABLE (
    disciplina VARCHAR,
    codigo_turma VARCHAR, 
    id_sala VARCHAR,      
    id_instituto INTEGER,
    hora_inicio TIME,
    hora_fim TIME
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        d.nome,
        oa.codigo_turma, 
        oa.id_sala,     
        i.id_instituto,
        ha.hora_inicio,
        ha.hora_fim
    FROM
        horario_aula ha
    JOIN
        oferta_aula oa ON ha.id_oferta = oa.id_oferta 
    JOIN
        disciplina d ON oa.sigla_disciplina = d.sigla
    JOIN
        departamento dep ON d.id_departamento = dep.id_departamento
    JOIN
        instituto i ON dep.id_instituto = i.id_instituto 
    WHERE
        ha.dia_semana ILIKE p_dia_semana
        AND oa.vigencia = p_vigencia
        AND i.id_instituto = p_id_instituto 
    ORDER BY
        ha.hora_inicio, d.nome;
END;
$$ LANGUAGE plpgsql;

--- 10. Negar Pedido de Reserva (Exemplo simples de mudança de status de reserva)
CREATE OR REPLACE FUNCTION negar_reserva(
    p_id_pedido INT,
    p_nusp_aprovador INT
) RETURNS VOID AS $$
BEGIN
    UPDATE pedido_reserva
    SET status = 3, -- 3 = Negado
        nusp_aprovador = p_nusp_aprovador,
        timestamp_veredito = NOW()
    WHERE id_pedido = p_id_pedido
      AND status = 1; -- Apenas se estiver pendente

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Pedido de reserva % não encontrado ou já processado.', p_id_pedido;
    END IF;

    RAISE NOTICE 'Pedido de reserva % foi negado.', p_id_pedido;
END;
$$ LANGUAGE plpgsql;

--- 11. Atualizar Detalhes do Recurso de uma Sala (Adiciona ou Remove| Operação frquente) 
CREATE OR REPLACE FUNCTION atualizar_recurso_sala_delta(
    p_id_sala VARCHAR,
    p_id_instituto INT, --- Encontrando salas 
    p_coluna_recurso VARCHAR, -- Ex: 'n_ar_cadeiras'
    p_delta INT -- Valor a adicionar/subtrair (+1, -1, etc.)
) RETURNS VOID AS $$
BEGIN
    -- Atualiza o recurso 
    EXECUTE format('
        UPDATE recursos
        SET %I = %I + $3
        WHERE id_sala = $1 AND id_instituto = $2;
    ', p_coluna_recurso, p_coluna_recurso)
    USING p_id_sala, p_id_instituto, p_delta;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Sala % no Instituto % não possui registro de recursos.', p_id_sala, p_id_instituto;
    END IF;
    
    RAISE NOTICE 'Recurso % atualizado por % na sala %.', p_coluna_recurso, p_delta, p_id_sala;
END;
$$ LANGUAGE plpgsql;

-- 1. Trigger para formatar a sigla da disciplina em MAIÚSCULA (Regra simples de formatação)

CREATE OR REPLACE FUNCTION trg_format_disciplina_sigla()
RETURNS TRIGGER AS $$
BEGIN
    NEW.sigla := UPPER(NEW.sigla);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

--- Dispara antes de qualquer insert ou update 
CREATE TRIGGER trg_upper_sigla
BEFORE INSERT OR UPDATE OF sigla ON disciplina FOR EACH ROW
	EXECUTE FUNCTION trg_format_disciplina_sigla();

-- 2. Trigger para checar se o aluno pode avaliar o professor (está matriculado na turma desse professor)
CREATE OR REPLACE FUNCTION trg_check_aluno_matriculado_avaliacao()
RETURNS TRIGGER AS $$
BEGIN
    IF NOT EXISTS (
        SELECT 1
        FROM participa_em AS p 
        JOIN leciona l ON p.sigla_disciplina = l.sigla_disciplina
                       AND p.codigo_turma = l.codigo_turma
                       AND p.vigencia = l.vigencia
        WHERE p.nusp_aluno = NEW.nusp_aluno
          AND l.nusp_professor = NEW.nusp_professor
          AND l.vigencia = NEW.vigencia 
    ) THEN
        RAISE EXCEPTION 'O aluno não está elegível para avaliar este professor nesta vigência.';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 3. Trigger para checar se o aluno pode avaliar o monitor (foi monitor da turma desse aluno)
CREATE OR REPLACE FUNCTION trg_check_aluno_matriculado_avaliacao_monitor()
RETURNS TRIGGER AS $$
BEGIN
    IF NOT EXISTS (
        SELECT 1
        FROM participa_em AS p 
        JOIN monitoria m ON p.sigla_disciplina = m.sigla_disciplina
                       AND p.codigo_turma = m.codigo_turma
                       AND p.vigencia = m.vigencia
        WHERE p.nusp_aluno = NEW.nusp_aluno
          AND m.nusp_aluno = NEW.nusp_monitor
          AND m.vigencia = NEW.vigencia 
    ) THEN
        RAISE EXCEPTION 'O aluno não está elegível para avaliar este monitor nesta vigência.';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

--- Dispara antes de qualquer insert 
CREATE TRIGGER trg_check_avaliacao_eligibility_mo
BEFORE INSERT ON avaliacao_professor FOR EACH ROW
EXECUTE FUNCTION trg_check_aluno_matriculado_avaliacao_monitor();

-- 4. Trigger para impedir que um aluno monitore uma turma em que ele está matriculado
CREATE OR REPLACE FUNCTION trg_check_monitor_matricula_conflito()
RETURNS TRIGGER AS $$
BEGIN
    IF EXISTS (
        SELECT 1
        FROM participa_em AS p
        WHERE p.nusp_aluno = NEW.nusp_monitor
          AND p.sigla_disciplina = NEW.sigla_disciplina
          AND p.codigo_turma = NEW.codigo_turma
          AND p.vigencia = NEW.vigencia
    ) THEN
        RAISE EXCEPTION 'O aluno monitor não pode estar matriculado na mesma turma que irá monitorar.';

	EXECUTE FUNCTION trg_check_monitor_matricula_conflito();

-- 6. Checa Pré-Requisitos antes da matrícula 
--- Crialção da função auxiliar 
CREATE OR REPLACE FUNCTION fn_checar_pre_requisito_aprovado(
    p_nusp_aluno INTEGER,
    p_sigla_disciplina VARCHAR
)
RETURNS BOOLEAN AS $$
DECLARE
    v_aprovado BOOLEAN;
BEGIN
	--- Seleciona se existe pelo menos um registro desse aluno em que ele não reprovou 
	--- Essa análise é suficiente pois permite analisar somente aluno e disciplina
	--- Pois, se em alguma vez ele foi aprovado, ele não fará mais
    SELECT EXISTS (
        SELECT 1
        FROM public.participa_em  
        WHERE nusp_aluno = p_nusp_aluno
          AND p.sigla_disciplina = p_sigla_disciplina
          AND media_final > 5.0 --- Requisitos usuais da USP
          AND frequencia > 0.70
		  
    )
    INTO v_aprovado;

    RETURN v_aprovado;
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION trg_check_prerequisitos_matricula()
RETURNS TRIGGER AS $$
DECLARE
    v_pre_req_sigla VARCHAR;
BEGIN
    FOR v_pre_req_sigla IN
        SELECT sigla_disciplina_requerida
        FROM requisito
        WHERE sigla_disciplina_requerente = NEW.sigla_disciplina
    LOOP
        IF NOT fn_checar_pre_requisito_aprovado(NEW.nusp_aluno, v_pre_req_sigla) THEN
            RAISE EXCEPTION 'Matrícula negada: O aluno não foi aprovado no pré-requisito: %.', v_pre_req_sigla;
        END IF;
    END LOOP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

--- Antes de matricular o aluno em uma nova turma, essa regra dispara 
CREATE TRIGGER trg_check_prerequisitos
	BEFORE INSERT ON participa_em FOR EACH ROW
	EXECUTE FUNCTION trg_check_prerequisitos_matricula();

-- 8. Garante que o RG seja armazenado apenas com números
CREATE OR REPLACE FUNCTION trg_clean_rg_number()
RETURNS TRIGGER AS $$
BEGIN
    NEW.rg := regexp_replace(NEW.rg, '[^0-9]', '', 'g'); -- Remove tudo que não for dígito
    IF LENGTH(NEW.rg) <> 9 THEN -- Assume 9 dígitos para o RG (char(9))
        RAISE EXCEPTION 'RG formatado é inválido (deve ter 9 dígitos).';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE TRIGGER trg_rg_cleaner
BEFORE INSERT OR UPDATE OF rg ON usuario FOR EACH ROW
	EXECUTE FUNCTION trg_clean_rg_number();

-- 9. Impede a matrícula em horários com conflito 
CREATE OR REPLACE FUNCTION prevenir_conflito_horario()
RETURNS TRIGGER AS $$
BEGIN
    -- Verifica se algum aluno matriculado na nova turma já tem aula no mesmo horário
    IF EXISTS (
        SELECT 1
        FROM participa_em p
        INNER JOIN oferta_aula oa_existente
            ON p.sigla_disciplina = oa_existente.sigla_disciplina
           AND p.codigo_turma    = oa_existente.codigo_turma
           AND p.vigencia        = oa_existente.vigencia
        INNER JOIN horario_aula ha_existente
            ON oa_existente.id_oferta = ha_existente.id_oferta

        INNER JOIN oferta_aula oa_nova
            ON NEW.id_oferta = oa_nova.id_oferta
        INNER JOIN participa_em p2
            ON p2.nusp_aluno       = p.nusp_aluno
           AND p2.sigla_disciplina = oa_nova.sigla_disciplina
           AND p2.codigo_turma     = oa_nova.codigo_turma
           AND p2.vigencia         = oa_nova.vigencia

        WHERE ha_existente.dia_semana = NEW.dia_semana
          AND ha_existente.hora_inicio < NEW.hora_fim
          AND ha_existente.hora_fim    > NEW.hora_inicio
          AND ha_existente.id_oferta  != NEW.id_oferta
    ) THEN
        RAISE EXCEPTION
            'Conflito de horário: Alunos matriculados já possuem aula neste horário';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

--- Antes de inserir ou atualizar 
CREATE TRIGGER trigger_prevenir_conflito_horario
BEFORE INSERT OR UPDATE ON horario_aula
FOR EACH ROW
EXECUTE FUNCTION prevenir_conflito_horario();

--- 10. Atualização do horário do veredito para o pedido de reserva 
CREATE OR REPLACE FUNCTION trg_update_veredito_timestamp()
RETURNS TRIGGER AS $$
BEGIN
    -- Se o status for alterado (de 1-pendente para 2-aprovado ou 3-negado)
    IF NEW.status <> OLD.status AND NEW.status IN (2, 3) THEN
        NEW.timestamp_veredito = NOW();
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Antes da atualização
CREATE OR REPLACE TRIGGER trg_set_veredito_time
BEFORE UPDATE OF status ON public.pedido_reserva FOR EACH ROW
    EXECUTE FUNCTION trg_update_veredito_timestamp();
