CREATE SCHEMA IF NOT EXISTS public;

-- Tabela instituto
DROP TABLE IF EXISTS public.instituto CASCADE;
CREATE TABLE IF NOT EXISTS public.instituto (
    id_instituto SERIAL,
    sigla varchar(10) UNIQUE NOT NULL,
    contato varchar(120) UNIQUE,
    email varchar(150),
    endereco varchar(100) UNIQUE,
    logradouro varchar(40),
    telefone varchar(20),
    cep varchar(8),
    PRIMARY KEY (id_instituto),
    CHECK (LENGTH(cep) = 8)
);

-- Tabela usuario
DROP TABLE IF EXISTS public.usuario CASCADE;
CREATE TABLE IF NOT EXISTS public.usuario (
    email varchar(150) NOT NULL UNIQUE,
    senha varchar(50) NOT NULL,
    nusp integer,
    rg char(9) UNIQUE,
    foto text,
    nome varchar(100),
    pnome varchar(50),
    unome varchar(50),
    PRIMARY KEY (nusp),
    CHECK (LENGTH(senha) >= 3),
    CHECK (nusp >= 0)
);

-- Tabela departamento
DROP TABLE IF EXISTS public.departamento CASCADE;
CREATE TABLE IF NOT EXISTS public.departamento (
    id_departamento SERIAL,
    nome varchar(100) UNIQUE NOT NULL,
    sigla varchar(20) UNIQUE NOT NULL,
    contato varchar(50) UNIQUE,
    email varchar(150),
    telefone varchar(20),
    id_instituto integer NOT NULL,
    PRIMARY KEY (id_departamento)
);

-- Tabela disciplina
DROP TABLE IF EXISTS public.disciplina CASCADE;
CREATE TABLE IF NOT EXISTS public.disciplina (
    sigla varchar(20),
    nome varchar(150) NOT NULL,
    objetivo varchar(500),
    conteudo varchar(500),
    bibliografia varchar(500),
    id_departamento integer NOT NULL,
    PRIMARY KEY (sigla)
);

-- Tabela curso
DROP TABLE IF EXISTS public.curso CASCADE;
CREATE TABLE IF NOT EXISTS public.curso (
    nome varchar(50) NOT NULL,
    modalidade varchar(20) NOT NULL,
    periodo varchar(20) NOT NULL ,
    id_curso SERIAL,
    carga_horaria integer,
    id_instituto integer NOT NULL,
    PRIMARY KEY (id_curso),
    CHECK (carga_horaria > 0),
    CHECK (modalidade IN ('presencial', 'EAD', 'híbrido')),
    CHECK (periodo IN ('matutino', 'vespertino', 'noturno', 'integral'))
);

-- Tabela grade
DROP TABLE IF EXISTS public.grade CASCADE;
CREATE TABLE IF NOT EXISTS public.grade (
    tipo_disciplina smallint NOT NULL,
    id_curso integer NOT NULL,
    sigla_disciplina varchar(20) NOT NULL,
    PRIMARY KEY (id_curso, sigla_disciplina),
    CHECK (tipo_disciplina IN (1, 2)) -- 1=obrigatória, 2=optativa
);

-- Tabela sala
DROP TABLE IF EXISTS public.sala CASCADE;
CREATE TABLE IF NOT EXISTS public.sala (
    id_sala varchar(4),
    lot_max integer NOT NULL,
    descricao varchar(240),
    id_instituto integer NOT NULL,
    PRIMARY KEY (id_sala, id_instituto),
    CHECK (lot_max > 0)
);

-- Tabela avaliacao_sala
DROP TABLE IF EXISTS public.avaliacao_sala CASCADE;
CREATE TABLE IF NOT EXISTS public.avaliacao_sala (
    time_stamp timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
    nota decimal NOT NULL,
    comentario varchar(240) NOT NULL,
    nusp_avaliador integer NOT NULL,
    id_sala varchar(4) NOT NULL,
    id_instituto integer NOT NULL,
    PRIMARY KEY (time_stamp, nusp_avaliador, id_sala, id_instituto),
    CHECK (nota >= 0 AND nota <= 10)
);

-- Tabela aluno
DROP TABLE IF EXISTS public.aluno CASCADE;
CREATE TABLE IF NOT EXISTS public.aluno (
    num_matricula integer NOT NULL UNIQUE,
    nusp_usuario integer,
    id_curso integer,
    PRIMARY KEY (nusp_usuario),
    CHECK (num_matricula > 0)
);

-- Tabela avaliacao_monitor
DROP TABLE IF EXISTS public.avaliacao_monitor CASCADE;
CREATE TABLE IF NOT EXISTS public.avaliacao_monitor (
    nota decimal NOT NULL,
    comentario varchar(240) NOT NULL,
    nusp_aluno integer NOT NULL,
    nusp_monitor integer NOT NULL,
    time_stamp timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (nusp_aluno, nusp_monitor, time_stamp),
    CHECK (nota >= 0 AND nota <= 10)
);

-- Tabela associado
DROP TABLE IF EXISTS public.associado CASCADE;
CREATE TABLE IF NOT EXISTS public.associado (
    sala varchar(15),
    num integer,
    bloco varchar(6),
    nusp_usuario integer,
    PRIMARY KEY (nusp_usuario)
);

-- Tabela vinculo_trabalho
DROP TABLE IF EXISTS public.vinculo_trabalho CASCADE;
CREATE TABLE IF NOT EXISTS public.vinculo_trabalho (
    cargo varchar(40) NOT NULL,
    data_inicio date NOT NULL,
    data_fim date,
    tipo_vinculo smallint NOT NULL,
    nusp_associado integer NOT NULL,
    id_instituto integer NOT NULL,
    PRIMARY KEY (nusp_associado, id_instituto),
    CHECK (data_fim IS NULL OR data_fim >= data_inicio),
    CHECK (tipo_vinculo IN (1, 2)) -- 1 Temporário, 2 Efetivado 
);

-- Tabela turma
DROP TABLE IF EXISTS public.turma CASCADE;
CREATE TABLE IF NOT EXISTS public.turma (
    sigla_disciplina varchar(20) NOT NULL,
    codigo_turma varchar(5) NOT NULL,
    vigencia varchar(10) NOT NULL,
    ano integer NOT NULL,
    semestre smallint NOT NULL,
    PRIMARY KEY (sigla_disciplina, codigo_turma, vigencia),
    CHECK (semestre IN (1, 2)),
    CHECK (ano >= 1900 AND ano <= 2100)
);

-- Tabela representa_disciplina
DROP TABLE IF EXISTS public.representa_disciplina CASCADE;
CREATE TABLE IF NOT EXISTS public.representa_disciplina (
    nusp_professor integer NOT NULL,
    sigla_disciplina varchar(20) NOT NULL,
    data_inicio date,
    data_fim date,
    PRIMARY KEY (nusp_professor, sigla_disciplina),
    CHECK (data_fim IS NULL OR data_fim >= data_inicio)
);

-- Tabela oferta_aula
DROP TABLE IF EXISTS public.oferta_aula CASCADE;
CREATE TABLE IF NOT EXISTS public.oferta_aula (
    id_oferta SERIAL,
    id_sala varchar(4),
    id_instituto integer,
    sigla_disciplina varchar(20) NOT NULL,
    codigo_turma varchar(5) NOT NULL,
    vigencia varchar(10) NOT NULL,
    PRIMARY KEY (id_oferta)
);

-- Tabela pedido_reserva
DROP TABLE IF EXISTS public.pedido_reserva CASCADE;
CREATE TABLE IF NOT EXISTS public.pedido_reserva (
    id_pedido SERIAL,
    proposito text NOT NULL,
    horario_geracao timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
    data_reserva date NOT NULL,
    hora_inicio_reserva time NOT NULL,
    hora_fim_reserva time NOT NULL,
    status smallint NOT NULL,
    timestamp_veredito timestamp,
    id_sala varchar(4) NOT NULL,
    id_instituto integer NOT NULL,
    nusp_solicitante integer NOT NULL,
    nusp_aprovador integer,
    PRIMARY KEY (id_pedido),
    CHECK (hora_fim_reserva > hora_inicio_reserva),
    CHECK (status IN (1, 2, 3)) -- 1=pendente, 2=aprovado, 3=rejeitado
);

-- Tabela coordenacao
DROP TABLE IF EXISTS public.coordenacao CASCADE;
CREATE TABLE IF NOT EXISTS public.coordenacao (

-- Tabela professor

-- Tabela participa_em

--- 1.Código para gerar a distribuição aleatória de alunos
DO $$
DECLARE
    r RECORD;
    curso_aleatorio INTEGER;
    matricula_aleatoria INTEGER;
BEGIN
    FOR r IN 
        SELECT nusp FROM usuario WHERE nusp BETWEEN 10000000 AND 99999999
    LOOP
        -- gera número de matrícula aleatório (6 dígitos)
        matricula_aleatoria := FLOOR(RANDOM() * 900000 + 100000);

        -- escolhe um curso aleatório
        SELECT id_curso INTO curso_aleatorio FROM curso ORDER BY RANDOM()
        LIMIT 1;

        -- insere o aluno
        INSERT INTO aluno (num_matricula, nusp_usuario, id_curso)
        VALUES (matricula_aleatoria, r.nusp, curso_aleatorio)
        ON CONFLICT (nusp_usuario) DO NOTHING;
    END LOOP;
END $$;

--- 3.Comando utilizado para aleatorização dos funcionários
DO $$
DECLARE
    num INTEGER;
    bloco VARCHAR(6);
    sala_assoc VARCHAR(10);
    nusp_associado INTEGER;
    i INTEGER := 1;
BEGIN
    WHILE i <= 230 LOOP
        LOOP
            -- gera número de sala aleatório (2 dígitos)
            num := FLOOR(RANDOM() * 900 + 100);

            -- gera bloco aleatório (A, B ou C)
            bloco := (ARRAY['A','B','C'])[FLOOR(RANDOM() * 3 + 1)];

            -- concatena bloco + número
            sala_assoc := bloco || num::TEXT;

            -- escolhe um nusp existente aleatório de 5 dígitos que ainda não está na tabela associado
            SELECT nusp INTO nusp_associado 
            FROM usuario
            WHERE nusp BETWEEN 10000 AND 99999
              AND nusp NOT IN (SELECT nusp_usuario FROM associado)
            ORDER BY RANDOM()
            LIMIT 1;

            -- se não houver NUSP disponível, sair do loop
            IF nusp_associado IS NOT NULL THEN
                -- verifica se a sala ainda não foi usada
                EXIT WHEN NOT EXISTS (
                    SELECT 1 
                    FROM associado
                    WHERE sala = sala_assoc
                );
            ELSE
                RAISE NOTICE 'Não há mais NUSPs disponíveis. Foram geradas % associações.', i-1;
                RETURN;
            END IF;
        END LOOP;

        -- insere a combinação única
        INSERT INTO associado (num, bloco, sala, nusp_usuario)
        VALUES (num, bloco, sala_assoc, nusp_associado);
